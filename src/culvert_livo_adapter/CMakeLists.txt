cmake_minimum_required(VERSION 3.16)
project(culvert_livo_adapter)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(tf2_ros REQUIRED)

add_executable(culvert_livo_adapter_node src/adapter.cpp)
target_compile_features(culvert_livo_adapter_node PRIVATE cxx_std_17)
target_compile_options(culvert_livo_adapter_node PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math)
target_include_directories(culvert_livo_adapter_node PRIVATE include)
target_link_libraries(culvert_livo_adapter_node Eigen3::Eigen)
ament_target_dependencies(culvert_livo_adapter_node rclcpp geometry_msgs sensor_msgs nav_msgs diagnostic_msgs tf2_ros)
install(TARGETS culvert_livo_adapter_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(adapter_math_test test/adapter_math_test.cpp)
  target_include_directories(adapter_math_test PRIVATE include)
  target_link_libraries(adapter_math_test Eigen3::Eigen)
  ament_target_dependencies(adapter_math_test sensor_msgs)
  target_compile_features(adapter_math_test PRIVATE cxx_std_17)
  target_compile_options(adapter_math_test PRIVATE -fno-fast-math)
endif()
ament_package()

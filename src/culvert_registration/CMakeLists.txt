cmake_minimum_required(VERSION 3.16)
project(culvert_registration LANGUAGES C CXX)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(PCL 1.12 REQUIRED COMPONENTS common io filters search registration)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
if(NOT TARGET JsonCpp::JsonCpp)
  find_package(jsoncpp CONFIG REQUIRED)
endif()
find_package(OpenMP REQUIRED COMPONENTS CXX)
find_package(small_gicp CONFIG REQUIRED)
add_library(culvert_registration_core STATIC src/session.cpp src/cloud.cpp src/evidence.cpp src/icp.cpp src/gicp.cpp)
target_compile_features(culvert_registration_core PUBLIC cxx_std_17)
target_compile_options(culvert_registration_core PRIVATE -Wall -Wextra)
target_include_directories(culvert_registration_core SYSTEM PUBLIC ${PCL_INCLUDE_DIRS})
target_include_directories(culvert_registration_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>
  $<INSTALL_INTERFACE:include/culvert_registration>)
target_link_libraries(culvert_registration_core PUBLIC ${PCL_LIBRARIES} Eigen3::Eigen JsonCpp::JsonCpp
  PRIVATE OpenSSL::Crypto OpenMP::OpenMP_CXX small_gicp::small_gicp)
add_executable(registration_server src/server.cpp)
target_compile_options(registration_server PRIVATE -Wall -Wextra)
target_link_libraries(registration_server PRIVATE culvert_registration_core)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  enable_testing()
  add_test(NAME registration_protocol COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/test/protocol.py $<TARGET_FILE:registration_server>)
  add_executable(checked_session_driver test/checked_driver.cpp)
  target_link_libraries(checked_session_driver PRIVATE culvert_registration_core)
  add_test(NAME registration_checked_binding COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/test/checked.py binding $<TARGET_FILE:checked_session_driver>)
  add_test(NAME registration_invalid_map_preserves_generation COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/test/checked.py invalid-maps $<TARGET_FILE:registration_server>)
endif()
install(TARGETS registration_server DESTINATION lib/${PROJECT_NAME})
install(TARGETS culvert_registration_core EXPORT registrationTargets ARCHIVE DESTINATION lib)
ament_export_targets(registrationTargets HAS_LIBRARY_TARGET)
install(FILES src/registration.hpp src/session.hpp src/checked_session.hpp DESTINATION include/${PROJECT_NAME})
install(DIRECTORY licenses DESTINATION share/${PROJECT_NAME})
ament_package(CONFIG_EXTRAS cmake/registration-dependencies.cmake)
